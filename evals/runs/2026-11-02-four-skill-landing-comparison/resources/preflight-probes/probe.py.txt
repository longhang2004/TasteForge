from pathlib import Path
from datetime import datetime, timezone
import hashlib
import json
import os
import subprocess
import tempfile
import urllib.request

repo = Path('/mnt/c/Users/hi/src/personal_repos/TasteForge')
run = repo / 'evals/runs/2026-11-02-four-skill-landing-comparison'
runtime = repo / '.tmp/comparison-runtime'
logs = run / 'resources/preflight-probes'
logs.mkdir()
now = lambda: datetime.now(timezone.utc).isoformat()
sha = lambda data: hashlib.sha256(data).hexdigest()
record = {'startedAt': now(), 'probes': [], 'runtime': {}, 'scope': 'Read-only/synthetic probes, explicit local paths, no host/global installation. Probes are environment checks, not UI-quality outcomes.'}
def command(name, args, cwd, env=None):
    row = {'name': name, 'startedAt': now(), 'command': args, 'cwd': str(cwd)}
    try:
        result = subprocess.run(args, cwd=cwd, env=env, capture_output=True, text=True, timeout=45)
        row.update({'exitCode': result.returncode, 'stdout': result.stdout, 'stderr': result.stderr})
    except Exception as exc:
        row.update({'exitCode': None, 'error': repr(exc)})
    row['finishedAt'] = now()
    (logs / (name + '.json')).write_text(json.dumps(row, indent=2) + '\n')
    record['probes'].append(row)
    print(name, row.get('exitCode'), row.get('error', ''), flush=True)
    return row

env = dict(os.environ)
env['PYTHONDONTWRITEBYTECODE'] = '1'
prodir = Path(tempfile.mkdtemp(prefix='tasteforge-promax-preflight-'))
pro_script = run / 'resources/promax/skill/scripts/search.py'
command('promax-design-system', ['python3', str(pro_script), 'community workshop landing', '--design-system', '-p', 'Preflight Fixture', '--json'], prodir, env)

skill = run / 'resources/impeccable/skill'
version = (skill / 'scripts/VERSION').read_text().strip()
lines = (skill / 'scripts/engine-checksums.txt').read_text().splitlines()
asset = 'impeccable-linux-x64'
expected = next(line.split()[2] for line in lines if line.startswith(version + ' ' + asset + ' '))
url = f'https://github.com/pbakaus/impeccable/releases/download/engine-v{version}/{asset}'
binary = runtime / 'bin' / version / asset
binary.parent.mkdir(parents=True, exist_ok=True)
download = {'version': version, 'asset': asset, 'url': url, 'path': str(binary), 'expectedSha256': expected, 'startedAt': now()}
try:
    if not binary.exists():
        data = urllib.request.urlopen(urllib.request.Request(url, headers={'User-Agent': 'TasteForge-resource-preflight'}), timeout=45).read()
        assert sha(data) == expected, 'Pinned engine checksum mismatch; do not execute'
        binary.write_bytes(data)
    data = binary.read_bytes()
    assert sha(data) == expected, 'Existing runtime checksum mismatch; do not execute'
    binary.chmod(0o755)
    download.update({'sha256': sha(data), 'bytes': len(data), 'status': 'verified'})
except Exception as exc:
    download.update({'status': 'unavailable', 'error': repr(exc)})
download['finishedAt'] = now()
record['runtime'] = download
(logs / 'impeccable-download.json').write_text(json.dumps(download, indent=2) + '\n')
if download['status'] == 'verified':
    env2 = dict(os.environ)
    env2['IMPECCABLE_BIN'] = str(binary)
    env2['IMPECCABLE_HOME'] = str(runtime / 'impeccable-home')
    env2['IMPECCABLE_SKILL_DIR'] = str(skill)
    command('impeccable-engine-handshake', ['sh', str(skill / 'scripts/impeccable'), 'engine-probe'], runtime, env2)
    command('impeccable-context-help', ['sh', str(skill / 'scripts/impeccable'), 'context', '--help'], runtime, env2)
    impdir = Path(tempfile.mkdtemp(prefix='tasteforge-impeccable-preflight-'))
    command('impeccable-empty-context', ['sh', str(skill / 'scripts/impeccable'), 'context', '--target', 'index.html'], impdir, env2)
record['finishedAt'] = now()
(logs / 'summary.json').write_text(json.dumps(record, indent=2) + '\n')
print('Complete', json.dumps({'runtime': download['status'], 'probeCount': len(record['probes'])}), flush=True)
